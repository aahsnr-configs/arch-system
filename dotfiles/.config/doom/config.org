(defun +org-src-eglot-setup-h ()
  "Enable Eglot in org-src-mode buffers that have tangle files.
This sets buffer-file-name to the tangle target so LSP can properly
identify the file and project context."
  (when (and (boundp 'org-src-mode)
             org-src-mode
             (or (derived-mode-p 'python-ts-mode)
                 (derived-mode-p 'python-mode)))
    ;; Get source block info safely
    (when-let* ((info (ignore-errors (org-babel-get-src-block-info)))
                (params (nth 2 info))
                (tangle-file (cdr (assq :tangle params))))

      ;; Only proceed if tangle is set to a real file (not "no" or "nil")
      (when (and tangle-file
                 (stringp tangle-file)
                 (not (member tangle-file '("no" "nil")))
                 (not (string-empty-p tangle-file)))

        ;; Resolve tangle file path relative to the org file
        (when-let* ((org-buffer (and (boundp 'org-src--beg-marker)
                                    org-src--beg-marker
                                    (marker-buffer org-src--beg-marker)))
                   (org-file (buffer-file-name org-buffer))
                   (org-dir (file-name-directory org-file))
                   (tangle-path (expand-file-name tangle-file org-dir)))

          ;; Create tangle file and directory if they don't exist
          (let ((tangle-dir (file-name-directory tangle-path)))
            (unless (file-directory-p tangle-dir)
              (make-directory tangle-dir t))
            (unless (file-exists-p tangle-path)
              (with-temp-file tangle-path
                (insert (format "# Tangled from: %s\n" org-file)))))

          ;; Set buffer-file-name so Eglot can identify the project/file
          (setq-local buffer-file-name tangle-path)

          ;; Disable problematic UI features in org-src buffers
          (setq-local eldoc-echo-area-use-multiline-p nil
                      eglot-autoshutdown nil)

          ;; Start Eglot
          (eglot-ensure))))))

(add-hook 'org-src-mode-hook #'+org-src-eglot-setup-h)

** Project-Aware Bibliography System
#+begin_src emacs-lisp
(defun +latex-find-project-bibliographies ()
  "Find all .bib files in project (root, references/, bib/, bibliography/)."
  (when (and (featurep 'projectile)
             (fboundp 'projectile-project-p)
             (projectile-project-p))
    (when-let ((root (projectile-project-root)))
      (let ((bib-files '())
            (search-dirs (list root
                              (expand-file-name "references/" root)
                              (expand-file-name "bib/" root)
                              (expand-file-name "bibliography/" root))))
        (dolist (dir search-dirs)
          (when (file-directory-p dir)
            (dolist (file (directory-files dir t "\\.bib\\'"))
              (push file bib-files))))
        (delete-dups (nreverse bib-files))))))

(defun +latex-setup-project-bibliography ()
  "Setup bibliography infrastructure for current project."
  (interactive)
  (if (and (featurep 'projectile)
           (fboundp 'projectile-project-p)
           (projectile-project-p))
      (let* ((root (projectile-project-root))
             (bib-dir (expand-file-name "references/" root))
             (bib-file (expand-file-name "references.bib" bib-dir)))
        (unless (file-directory-p bib-dir)
          (make-directory bib-dir t))
        (unless (file-exists-p bib-file)
          (with-temp-file bib-file
            (insert "% Bibliography for "
                    (file-name-nondirectory (directory-file-name root)) "\n")
            (insert "% Created: " (format-time-string "%Y-%m-%d") "\n\n")))
        (message "Bibliography setup complete: %s" bib-file)
        bib-file)
    (message "Not in a project")))
