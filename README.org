#+TITLE: Arch Setup
* Custom Arch Linux Installation Guide 
This guide creates a custom Arch Linux installation that replicates
CachyOS features using:

- Full disk encryption (LUKS2 on LVM)
- Optimized Btrfs with extensive subvolume layout
- Dracut for initramfs generation (required for bootable btrfs
  snapshots)
- Limine bootloader (faster than GRUB, especially with encryption)
- Snapper with bootable snapshots via limine-snapper-sync
- Automatic snapshot creation before system updates
- CachyOS optimized repositories and kernel

** Prerequisites
- Boot into Arch Linux live environment
- Working network connection
- UEFI system (not BIOS/Legacy)

** Disk Partitioning and Encryption Setup
*** Partition the Disk
#+begin_src sh
cfdisk /dev/nvme0n1
#+end_src

Create two partitions:

- =/dev/nvme0n1p1= - *2GB*, type: EFI System (ESP - needs space for
  multiple kernels + snapshots)
- =/dev/nvme0n1p2= - Remaining space, type: Linux LVM

*** Create and Configure Encrypted LVM
#+begin_src sh
# Format ESP
mkfs.vfat -F 32 /dev/nvme0n1p1

# Create LUKS2 encrypted container
cryptsetup --cipher aes-xts-plain64 \
           --hash sha512 \
           --use-random \
           --verify-passphrase \
           luksFormat /dev/nvme0n1p2

# Open the encrypted container
cryptsetup luksOpen /dev/nvme0n1p2 cryptlvm

# Create physical volume and volume group
pvcreate /dev/mapper/cryptlvm
vgcreate vg0 /dev/mapper/cryptlvm

# Create logical volumes
lvcreate -L 24G vg0 -n swap
lvcreate -l 100%FREE vg0 -n root

# Format volumes
mkfs.btrfs -f /dev/vg0/root
mkswap /dev/vg0/swap

# Initial mount
mount /dev/vg0/root /mnt
swapon /dev/vg0/swap
#+end_src

*** Mount Subvolumes with Optimal Options
#+begin_src sh
# Mount options for performance and compression (matching CachyOS defaults)
BTRFS_OPTS="noatime,compress=zstd:3,space_cache=v2,discard=async"

# Mount root subvolume
mount -o ${BTRFS_OPTS},subvol=@ /dev/vg0/root /mnt

# Create mount points and mount all subvolumes
mkdir -p /mnt/{home,opt,tmp,root,srv,nix,usr/local,var/{cache/pacman/pkg,crash,tmp,spool,log/audit},.snapshots,boot}

mount -o ${BTRFS_OPTS},subvol=@home /dev/vg0/root /mnt/home
mount -o ${BTRFS_OPTS},subvol=@opt /dev/vg0/root /mnt/opt
mount -o ${BTRFS_OPTS},subvol=@tmp /dev/vg0/root /mnt/tmp
mount -o ${BTRFS_OPTS},subvol=@root /dev/vg0/root /mnt/root
mount -o ${BTRFS_OPTS},subvol=@srv /dev/vg0/root /mnt/srv
mount -o ${BTRFS_OPTS},subvol=@nix /dev/vg0/root /mnt/nix
mount -o ${BTRFS_OPTS},subvol=@usr@local /dev/vg0/root /mnt/usr/local
mount -o ${BTRFS_OPTS},subvol=@var /dev/vg0/root /mnt/var
mount -o ${BTRFS_OPTS},subvol=@var@cache /dev/vg0/root /mnt/var/cache
mount -o ${BTRFS_OPTS},subvol=@pkg /dev/vg0/root /mnt/var/cache/pacman/pkg
mount -o ${BTRFS_OPTS},subvol=@var@crash /dev/vg0/root /mnt/var/crash
mount -o ${BTRFS_OPTS},subvol=@var@tmp /dev/vg0/root /mnt/var/tmp
mount -o ${BTRFS_OPTS},subvol=@var@spool /dev/vg0/root /mnt/var/spool
mount -o ${BTRFS_OPTS},subvol=@var@log /dev/vg0/root /mnt/var/log
mount -o ${BTRFS_OPTS},subvol=@var@log@audit /dev/vg0/root /mnt/var/log/audit
mount -o ${BTRFS_OPTS},subvol=@snapshots /dev/vg0/root /mnt/.snapshots

# Enable swap

** Configure Dracut for LUKS+LVM+Btrfs
*** Get Your UUIDs
#+begin_src sh
# Get LUKS UUID (the partition, not the mapper)
LUKS_UUID=$(cryptsetup luksDump /dev/nvme0n1p2 | grep "UUID:" | head -1 | awk '{print $2}')
echo "LUKS UUID: $LUKS_UUID"

# Get root filesystem UUID (the mapper device)
ROOT_UUID=$(blkid -s UUID -o value /dev/mapper/vg0-root)
echo "Root UUID: $ROOT_UUID"

# Get swap UUID
SWAP_UUID=$(blkid -s UUID -o value /dev/mapper/vg0-swap)
echo "Swap UUID: $SWAP_UUID"

# Save these for later
echo "LUKS_UUID=$LUKS_UUID" > /tmp/uuids.txt
echo "ROOT_UUID=$ROOT_UUID" >> /tmp/uuids.txt
echo "SWAP_UUID=$SWAP_UUID" >> /tmp/uuids.txt
cat /tmp/uuids.txt
#+end_src

*** Create Dracut Configuration Files
*Main dracut configuration:*

#+begin_src sh
cat > /etc/dracut.conf.d/custom.conf << 'EOF'
# Host-only mode for smaller, faster initramfs
hostonly="yes"

# Let dracut auto-detect and embed kernel parameters from system config
hostonly_cmdline="yes"

# Compression
compress="zstd"

# Add required modules for LUKS + LVM + Btrfs + Resume
add_dracutmodules+=" crypt dm lvm rootfs-block resume "

# Omit unnecessary modules
omit_dracutmodules+=" network cifs nfs nbd brltty "

# Force inclusion of btrfs driver
force_drivers+=" btrfs "

# Use fstab for mounting
use_fstab="yes"

# Show modules during build (helpful for debugging)
show_modules="yes"
EOF
#+end_src

*** Configure /etc/crypttab
This is CRITICAL for dracut to properly unlock LUKS:

#+begin_src sh
# Create crypttab entry
# Format: mapper-name UUID=<uuid> none luks
cat > /etc/crypttab << EOF
# <name>    <device>                                  <password>  <options>
cryptlvm    UUID=${LUKS_UUID}                         none        luks,discard
EOF

# Verify
cat /etc/crypttab
#+end_src

*** Configure Limine (/etc/default/limine)
This is where you set kernel parameters for dracut:

#+begin_src sh
# The variables from section 5.1 are still active in this shell session
# Verify they are set
echo "LUKS UUID: $LUKS_UUID"
echo "Root UUID: $ROOT_UUID"
echo "Swap UUID: $SWAP_UUID"

# If variables are not set, source them:
# source /tmp/uuids.txt

# Create Limine configuration with dracut syntax
cat > /etc/default/limine << EOF
# ESP Path
ESP_PATH="/boot"

# Kernel command line parameters (dracut syntax!)
# Note: With hostonly_cmdline="yes", dracut auto-detects parameters from
# /etc/crypttab and /etc/fstab, but we specify them here for explicit control

# Basic boot parameters
KERNEL_CMDLINE[default]+="quiet loglevel=3 rw"

# Btrfs root subvolume
KERNEL_CMDLINE[default]+="rootflags=subvol=/@"

# LUKS unlock parameters
KERNEL_CMDLINE[default]+="rd.luks.uuid=${LUKS_UUID}"
KERNEL_CMDLINE[default]+="rd.luks.name=${LUKS_UUID}=cryptlvm"

# LVM activation (uses VG/LV names, NOT UUIDs)
KERNEL_CMDLINE[default]+="rd.lvm.lv=vg0/root"
KERNEL_CMDLINE[default]+="rd.lvm.lv=vg0/swap"

# Root filesystem (uses UUID - recommended by dracut documentation)
KERNEL_CMDLINE[default]+="root=UUID=${ROOT_UUID}"

# Resume from swap (uses UUID - recommended)
KERNEL_CMDLINE[default]+="resume=UUID=${SWAP_UUID}"

# Enable TRIM/discard for SSD on LUKS
KERNEL_CMDLINE[default]+="rd.luks.allow-discards"

# Boot order (* means all main kernels)
BOOT_ORDER="*, *lts, *fallback, Snapshots"

# Commands to run before/after saving config
# These ensure Limine bootloader files are properly enrolled
COMMANDS_BEFORE_SAVE="limine-reset-enroll"
COMMANDS_AFTER_SAVE="limine-enroll-config"
EOF

# Verify configuration was created correctly
cat /etc/default/limine
#+end_src

** Setup Snapper for Bootable Snapshots
Now that you're in a running system, set up Snapper properly.

*** Install Snapper and Dependencies
#+begin_src sh
sudo pacman -S snapper inotify-tools
#+end_src

*** Configure Snapper
#+begin_src sh
# Unmount the snapshots directory
sudo umount /.snapshots

# Remove the directory
sudo rm -rf /.snapshots

# Create Snapper configuration for root
sudo snapper -c root create-config /

# Snapper creates a .snapshots subvolume, but we want to use ours
# Delete Snapper's auto-created subvolume
sudo btrfs subvolume delete /.snapshots

# Recreate directory and remount our @snapshots subvolume
sudo mkdir /.snapshots
sudo mount /.snapshots

# Verify it's mounted
mount | grep snapshots

# Set proper permissions
sudo chmod 750 /.snapshots
sudo chown root:root /.snapshots

# Get the ID of @ subvolume and set it as default
ROOT_SUBVOL_ID=$(sudo btrfs subvolume list / | grep "path @$" | awk '{print $2}')
echo "Root subvolume ID: $ROOT_SUBVOL_ID"
sudo btrfs subvolume set-default ${ROOT_SUBVOL_ID} /
#+end_src

*Method 4: Manual Btrfs (Most Control)*

#+begin_src sh
# Boot into snapshot first, then:
sudo mkdir -p /mnt/btrfs_root
sudo mount -o subvolid=5 /dev/mapper/vg0-root /mnt/btrfs_root

# Delete current @ subvolume
sudo btrfs subvolume delete /mnt/btrfs_root/@

# Create new @ from snapshot (replace 123 with snapshot ID)

# Set as default
NEW_ID=$(sudo btrfs subvolume list /mnt/btrfs_root | grep "path @$" | awk '{print $2}')

# Clean up and reboot

* Chrooting into Existing Installation
This section provides instructions for accessing your installed system
from a live Arch environment for maintenance, recovery, or
troubleshooting purposes.

** When You Need This
- System won't boot and needs repair
- Forgot to install a critical package
- Need to regenerate initramfs after changes
- Kernel parameter adjustments
- Bootloader configuration fixes
- General system maintenance

** Prerequisites
- Bootable Arch Linux live USB
- Your LUKS encryption passphrase
- Network connection (if you need to download packages)
  
** Step-by-Step Chroot Process
*** Step 1: Boot into Arch Live Environment
Boot from your Arch Linux USB and wait for the live environment to load.

*** Step 2: Unlock the LUKS Encrypted Partition
#+begin_src sh
# Unlock the LUKS container (you'll be prompted for your passphrase)
cryptsetup luksOpen /dev/nvme0n1p2 cryptlvm

# Verify it was unlocked successfully
ls -la /dev/mapper/
# You should see: cryptlvm
#+end_src

*** Step 3: Activate LVM Volumes
#+begin_src sh
# Scan for volume groups
vgscan

# Activate all volume groups
vgchange -ay

# Verify volumes are active
lvs
# You should see vg0/root and vg0/swap
#+end_src

*** Step 4: Mount Btrfs Root Subvolume
#+begin_src sh
# Mount options (same as installation)
BTRFS_OPTS="noatime,compress=zstd:3,space_cache=v2,discard=async"

# Mount the root @ subvolume first
# This makes all the directories (/home, /var, /boot, etc.) available
mount -o ${BTRFS_OPTS},subvol=@ /dev/mapper/vg0-root /mnt

# Verify the directories exist
ls -la /mnt
# You should see: bin, boot, dev, etc, home, opt, root, srv, tmp, usr, var, etc.
#+end_src

*** Step 5: Mount All Other Subvolumes
*Important:* The directories already exist after mounting @ - you do NOT
need to create them!

#+begin_src sh
# Mount all subvolumes to their existing directories
mount -o ${BTRFS_OPTS},subvol=@home /dev/mapper/vg0-root /mnt/home
mount -o ${BTRFS_OPTS},subvol=@opt /dev/mapper/vg0-root /mnt/opt
mount -o ${BTRFS_OPTS},subvol=@tmp /dev/mapper/vg0-root /mnt/tmp
mount -o ${BTRFS_OPTS},subvol=@root /dev/mapper/vg0-root /mnt/root
mount -o ${BTRFS_OPTS},subvol=@srv /dev/mapper/vg0-root /mnt/srv
mount -o ${BTRFS_OPTS},subvol=@nix /dev/mapper/vg0-root /mnt/nix
mount -o ${BTRFS_OPTS},subvol=@usr@local /dev/mapper/vg0-root /mnt/usr/local
mount -o ${BTRFS_OPTS},subvol=@var /dev/mapper/vg0-root /mnt/var
mount -o ${BTRFS_OPTS},subvol=@var@cache /dev/mapper/vg0-root /mnt/var/cache
mount -o ${BTRFS_OPTS},subvol=@pkg /dev/mapper/vg0-root /mnt/var/cache/pacman/pkg
mount -o ${BTRFS_OPTS},subvol=@var@crash /dev/mapper/vg0-root /mnt/var/crash
mount -o ${BTRFS_OPTS},subvol=@var@tmp /dev/mapper/vg0-root /mnt/var/tmp

#+end_src

** Key Difference from Installation
*During installation:* You create empty subvolumes, then manually create
all directories with =mkdir -p= before mounting.

* Firewalld Setup 
This guide provides complete instructions for setting up and configuring
firewalld on Arch Linux to protect your system with a modern, dynamic
firewall.

** What is Firewalld?
Firewalld provides a dynamically managed firewall with support for
network/firewall zones that define the trust level of network
connections or interfaces. It has support for IPv4, IPv6 firewall
settings, ethernet bridges and IP sets. The Arch Wiki states that there
is a separation of runtime and permanent configuration options, and it
provides an interface for services or applications to add firewall rules
directly.

*Key advantages:* - Dynamic management (no need to restart) - Zone-based
configuration (easy to manage different network trust levels) - Uses
nftables backend (modern and efficient) - Better integration with
NetworkManager - Rich language for complex rules

** Installation
*** Install Firewalld
#+begin_src sh
# Install firewalld
sudo pacman -S firewalld

# Enable and start the service
sudo systemctl enable --now firewalld.service

# Verify it's running

sudo firewall-cmd --state
# Should output: running
#+end_src

*** Verify Configuration
#+begin_src sh

# Should output: public (by default)

# List all zones
sudo firewall-cmd --get-zones

# Check active zones
sudo firewall-cmd --get-active-zones

# List all available services
sudo firewall-cmd --get-services
#+end_src

** Understanding Zones
Firewalld uses zones to define trust levels for network connections.
Each zone has default rules that determine what traffic is allowed.

*** Pre-defined Zones
*From least to most trusted:*

1. *drop* - Drops all incoming packets without reply. Only outgoing
   connections are allowed.

2. *block* - Rejects all incoming connections with icmp-host-prohibited
   message.

3. *public* (default) - For use in public areas. You don't trust other
   computers. Only selected incoming connections are accepted. Default
   allows SSH and DHCPv6-client.

4. *external* - For use on external networks with masquerading (NAT)
   enabled. Only selected incoming connections are accepted.

5. *dmz* - For computers in your DMZ that have limited access to your
   internal network. Only selected incoming connections are accepted.

6. *work* - For use at work where you trust most computers. More
   services may be allowed.

7. *home* - For use at home where you trust most devices. More services
   may be allowed.

8. *internal* - For use on internal networks when you trust the other
   computers. More services are allowed.

9. *trusted* - All network connections are accepted. Complete trust.

*** Zone Best Practices
#+begin_src sh
# For desktop/laptop - use 'public' or 'home' depending on network
# For servers - use 'public' and explicitly allow only needed services
# For trusted internal network - use 'internal' or 'home'
#+end_src

** Part 15: Security Best Practices
1. *Default deny policy* - Only allow what's needed
2. *Use specific zones* - Don't put everything in ‘public’
3. *Regular audits* - Review rules monthly
4. *Least privilege* - Only open ports that must be accessible
5. *Source restrictions* - Use rich rules to limit by IP when possible
6. *Rate limiting* - Use rich rules with limits for brute-force
   protection
7. *Logging* - Enable for security-critical services
8. *Backup* - Keep configuration backups
9. *Test changes* - Test in runtime before making permanent
10. *Integration* - Combine with fail2ban for additional protection
